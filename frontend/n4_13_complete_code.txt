interface N4_13Props {
  data: AuditReportData;
}

export const N4_13: React.FC<N4_13Props> = ({ data }) => {
  const parseValue = (val: string | number): number => {
    if (!val || val === '' || val === '-') return 0;
    const str = typeof val === 'string' ? val.replace(/,/g, '').replace(/%/g, '').replace(/\(/g, '-').replace(/\)/g, '') : val.toString();
    const parsed = parseFloat(str);
    return isNaN(parsed) ? 0 : parsed;
  };

  const formatDisplay = (num: number): string => {
    if (num === 0) return '-';
    const absNum = Math.abs(num);
    const formatted = absNum.toLocaleString('en-US');
    return num < 0 ? `(${formatted})` : formatted;
  };

  const ppeTotals = data.ppe.assets.reduce((acc, asset) => {
    const co = parseValue(asset.costOpening);
    const ca = parseValue(asset.costAddition);
    const cd = parseValue(asset.costDisposal);
    const cc = co + ca - cd;

    const do_ = parseValue(asset.depOpening);
    const dc = parseValue(asset.depCharged);
    const da = parseValue(asset.depAdjustment);
    const dcl = do_ + dc + da;

    return {
      costOpening: acc.costOpening + co,
      costClosing: acc.costClosing + cc,
      depOpening: acc.depOpening + do_,
      depCharged: acc.depCharged + dc,
      depClosing: acc.depClosing + dcl,
    };
  }, { costOpening: 0, costClosing: 0, depOpening: 0, depCharged: 0, depClosing: 0 });

  const adminDep = parseValue(data.ppe.breakdown.adminExpense);
  const factoryDep = ppeTotals.depCharged - adminDep;

  const renderNoteHeader = (num: string, title: string) => (
    <div className="flex gap-2 font-bold mt-6 mb-2">
      <span className="w-12">{num}</span>
      <span>. {title}</span>
    </div>
  );

  const renderRow = (label: string, val25: string = '-', val24: string = '-', isBold: boolean = false, isItalic: boolean = false, indent: boolean = true) => (
    <div className={`flex justify-between py-0.5 ${isBold ? 'font-bold' : ''} ${isItalic ? 'italic' : ''}`}>
      <span className={`${indent ? 'pl-14' : 'pl-0'} flex-1`}>{label}</span>
      <div className="flex gap-8 text-right">
        <span className="w-24">{val25}</span>
        <span className="w-24">{val24}</span>
      </div>
    </div>
  );

  const renderTotalLine = () => (
    <div className="flex justify-end mt-1">
      <div className="flex gap-8">
        <div className="w-24 border-t border-black"></div>
        <div className="w-24 border-t border-black"></div>
      </div>
    </div>
  );

  const renderDoubleTotalLine = (label?: string) => (
    <div className="flex justify-end mt-0.5 relative">
      <div className="flex gap-8">
        <div className="w-24 border-t-4 border-double border-black"></div>
        <div className="w-24 border-t-4 border-double border-black"></div>
      </div>
      {label && (
        <div className="absolute left-[calc(100%+12px)] bottom-0 whitespace-nowrap font-serif text-[11px]">
          {label}
        </div>
      )}
    </div>
  );

  const contextValue = {
    data,
    formatDisplay,
    parseValue,
    renderNoteHeader,
    renderRow,
    renderTotalLine,
    renderDoubleTotalLine
  };

  return (
    <N4_13Context.Provider value={contextValue}>
      <div className="w-full bg-white p-8 font-serif text-[11px] leading-tight text-black">
        <div className="max-w-4xl mx-auto border border-blue-800 p-1 min-h-full">
          <div className="min-w-[600px]">
            {/* Header Section */}
          <div className="border-b-2 border-blue-900 mb-1 pb-1">
            <h1 className="font-bold text-base">{data.company}</h1>
            <p className="text-xs">{data.addr}</p>
          </div>

          <div className="flex justify-end mb-4">
            <div className="text-right">
              <p className="font-bold border-b border-black inline-block px-8">In Bangladesh Taka</p>
              <div className="flex gap-8 mt-1">
                <span className="w-24 text-center font-bold border-b border-black">{getFormattedDate(data.reportingDate)}</span>
                <span className="w-24 text-center font-bold border-b border-black">{getFormattedDate(data.startDate)}</span>
              </div>
            </div>
          </div>

          <Note4 ppeTotals={ppeTotals} />
          <Note5 />
          <Note6 />
          <Note7 />
          <Note8 />
          <Note9 />
          <Note10 />
          <Note11 />

          <Note25 adminDep={adminDep} />

import React, { createContext, useContext } from 'react';
import { AuditReportData } from '../../../types';

export interface N4_13ContextType {
  data: AuditReportData;
  formatDisplay: (num: number) => string;
  parseValue: (val: string | number) => number;
  renderNoteHeader: (num: string, title: string) => React.ReactNode;
  renderRow: (label: string, val25?: string, val24?: string, isBold?: boolean, isItalic?: boolean, indent?: boolean) => React.ReactNode;
  renderTotalLine: () => React.ReactNode;
  renderDoubleTotalLine: (label?: string) => React.ReactNode;
}

export const N4_13Context = createContext<N4_13ContextType | null>(null);

export const useN4_13 = () => {
  const context = useContext(N4_13Context);
  if (!context) throw new Error('useN4_13 must be used within N4_13Provider');
  return context;
};

export const TableHeader: React.FC<{ children: React.ReactNode; className?: string }> = ({ children, className = '' }) => (
  <th className={`border border-black py-1 px-2 text-center font-bold ${className}`}>{children}</th>
);

export const TableCell: React.FC<{ children: React.ReactNode; className?: string; colSpan?: number }> = ({ children, className = '', colSpan }) => (
  <td className={`border border-black py-1 px-2 ${className}`} colSpan={colSpan}>{children}</td>
);
